-- 1. Üye 9'un başlangıç bakiyesini sorgulayıp 40 kredi ekliyoruz ve COMMIT ediyoruz:
vtysdb=> SELECT kredi FROM krediler WHERE uye_id = 9;
 kredi 
-------
   270
(1 row)

vtysdb=> BEGIN;
BEGIN
vtysdb=*> UPDATE krediler SET kredi = kredi + 40 WHERE uye_id = 9;
UPDATE 1
vtysdb=*> COMMIT;
COMMIT

vtysdb=> SELECT kredi FROM krediler WHERE uye_id = 9;
 kredi 
-------
   310
(1 row)

-- 2. Sunucu yeniden başlatıldı:
-- $ docker compose restart postgres
-- Container vtys-postgres Restarting
-- Container vtys-postgres Started

-- 3. Yeniden bağlandıktan sonra bakiyeyi tekrar kontrol ediyoruz:
vtysdb=> SELECT kredi FROM krediler WHERE uye_id = 9;
 kredi 
-------
   310
(1 row)

-- Gözlem:
-- Sunucu yeniden başlatılmasına rağmen COMMIT edilen 310 bakiyesi kaybolmamıştır.
-- Bu durum ACID'in Durability (Dayanıklılık) özelliğidir ve PostgreSQL'de
-- Write-Ahead Logging (WAL) mekanizması sayesinde sağlanır.
